작동 과정 설명

  1. 게임용 이미지, 사운드의 링크를 클릭하여, images와 sounds 폴더를 다운로드 한다.
  1. 파일탐색기를 연다. C\ → 사용자 → 본인좌석번호 폴더 이동
  1. images , sounds 폴더 게임 소스코드(.py) 파일을 복사한다.
  1. 하단의 pygame 구동하기의 단계를 차례로 진행한다.

anaconda Command Prompt로 pygame 구동하기

(1) conda init
: Anaconda를 터미널(Command Prompt)에서 정상적으로 인식하고 사용할 수 있도록 초기화(준비)하는 명령어

(2) conda activate
: 기본 가상환경(base)을 활성화하는 명령어

(3) conda create –n pygame_env python=3.10
: pygame_env라는 이름의 가상환경을 만드는 명령어

(4) conda activate pygame_env
: (3)번에서 만든 pygame_env 가상환경(방) 안으로 들어가는 명령어

(5) pip install pygame
: 활성화된 가상환경 안에서 파이썬 패키지 관리자인 pip를 사용해 pygame 라이브러리를 다운로드하고 설치하는 명령어

(6) python umbrella.py (umbrella = 게임 이름)
: 파이썬을 이용해 내가 작성한 tank.py라는 이름의 게임 소스코드 파일을 실행하라는 명령어

전체 소스 코드(racing_car.py)

import sys
import pygame
import random

# 게임 화면 크기
WINDOW_WIDTH = 550
WINDOW_HEIGHT = 800

# 색상
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
GRAY = (150, 150, 150)

# 소스 디렉토리
DIRCARS = "cars/"
DIRSOUND = "sounds/"  # 사운드 폴더명

# 기본 전역 변수 초기화
STAGE = 1
CAR_COUNT = 5
SCORE = 0
STAGESCORE = 0
STAGESTAIR = 1000
PNUMBER = 5  # 플레이어 Life 갯수

# 자동차 오브젝트를 저장할 List
CARS = []
SCREEN = None  # main에서 초기화
LIFE_IMAGE = None  # 라이프 표시용 이미지를 미리 저장할 전역 변수


class Car:
    car_image = [
        'Car01.png', 'Car02.png', 'Car03.png', 'Car04.png', 'Car05.png',
        'Car06.png', 'Car07.png', 'Car08.png', 'Car09.png', 'Car10.png',
        'Car11.png', 'Car12.png', 'Car13.png', 'Car14.png', 'Car15.png',
        'Car16.png', 'Car17.png', 'Car18.png', 'Car19.png', 'Car20.png',
        'Car21.png', 'Car22.png', 'Car23.png', 'Car24.png', 'Car25.png',
        'Car26.png', 'Car27.png', 'Car28.png', 'Car29.png', 'Car30.png',
        'Car31.png', 'Car32.png', 'Car33.png', 'Car34.png', 'Car35.png',
        'Car36.png', 'Car37.png', 'Car38.png', 'Car39.png', 'Car40.png'
    ]
   
    def __init__(self, x=0, y=0, dx=0, dy=0):
        self.image = None
        self.x = x
        self.y = y
        self.dx = dx
        self.dy = dy
        self.rect = None
       
    def load_car(self, p=""):
        if p == "p":
            # 플레이어 차량
            self.image = pygame.image.load(DIRCARS + "Player.png")
            self.image = pygame.transform.scale(self.image, (40, 102))
            self.rect = self.image.get_rect()
            self.rect.x = self.x
            self.rect.y = self.y
        else:
            # 상대방 자동차
            self.image = pygame.image.load(DIRCARS + random.choice(self.car_image))
            self.rect = self.image.get_rect()
           
            if self.rect.width <= 55:
                carwidth = self.rect.width - 15
                carheight = round((self.rect.height * carwidth) / self.rect.width)
            else:
                carwidth = self.rect.width
                carheight = self.rect.height
               
            self.image = pygame.transform.scale(self.image, (carwidth, carheight))
            self.rect.width = carwidth
            self.rect.height = carheight
           
            self.rect.x = random.randrange(0, WINDOW_WIDTH - self.rect.width)
            self.rect.y = random.randrange(-150, -50)
           
            speed = STAGE + 5
            if speed > 15:
                speed = 15
            self.dy = random.randint(5, speed)
       
    def draw_car(self):
        SCREEN.blit(self.image, [self.rect.x, self.rect.y])

    def move_x(self):
        self.rect.x += self.dx

    def move_y(self):
        self.rect.y += self.dy
       
    def check_screen(self):
        if self.rect.left < 0:
            self.rect.left = 0
        if self.rect.right > WINDOW_WIDTH:
            self.rect.right = WINDOW_WIDTH
        if self.rect.top < 0:
            self.rect.top = 0
        if self.rect.bottom > WINDOW_HEIGHT:
            self.rect.bottom = WINDOW_HEIGHT
           
    def check_collision(self, car, distance=0):
        if (self.rect.top + distance < car.rect.bottom) and \
           (car.rect.top < self.rect.bottom - distance) and \
           (self.rect.left + distance < car.rect.right) and \
           (car.rect.left < self.rect.right - distance):
            return True
        else:
            return False

def increase_score():
    global SCORE, STAGE, STAGESCORE
    SCORE += 10
   
    if STAGE == 1:
        stair = STAGESTAIR
    else:
        stair = (STAGE - 1) * STAGESTAIR
       
    if SCORE >= STAGESCORE + stair:
        STAGE += 1
        STAGESCORE = STAGESCORE + stair

def draw_score():
    font_01 = pygame.font.SysFont("FixedSys", 30, True, False)
   
    # SCORE 기록
    text_score = font_01.render("Score : " + str(SCORE), True, BLACK)
    SCREEN.blit(text_score, [15, 15])

    # STAGE 기록
    text_stage = font_01.render("STAGE : " + str(STAGE), True, BLACK)
    text_stage_rect = text_stage.get_rect()
    text_stage_rect.centerx = round(WINDOW_WIDTH / 2)
    SCREEN.blit(text_stage, [text_stage_rect.x, 15])

    # 플레이어 Life 기록
    global PNUMBER, LIFE_IMAGE
    
    # [수정] 5개까지만 이미지를 그리고, 초과분은 반복문 밖에서 한 번만 텍스트 처리
    display_lives = min(PNUMBER, 5)
    for i in range(display_lives):
        if LIFE_IMAGE:
            px = WINDOW_WIDTH - 20 - (i * 30)
            SCREEN.blit(LIFE_IMAGE, [px, 15])
            
    if PNUMBER > 5:
        text_pnumber = font_01.render("+" + str(PNUMBER - 5), True, BLACK)
        text_pnumber_x = WINDOW_WIDTH - 30 - (5 * 30)
        SCREEN.blit(text_pnumber, [text_pnumber_x, 15])

def main():
    global SCREEN, CAR_COUNT, WINDOW_WIDTH, WINDOW_HEIGHT, PNUMBER, LIFE_IMAGE

    pygame.init()
    SCREEN = pygame.display.set_mode((WINDOW_WIDTH, WINDOW_HEIGHT),
                                     pygame.HWSURFACE | pygame.DOUBLEBUF | pygame.RESIZABLE)
   
    pygame.display.set_caption("Racing Car Game")
   
    # [수정] NameError 방지를 위해 사운드 변수 미리 초기화
    sound_crash = None
    mixer_initialized = False
    
    try:
        pygame.mixer.init()
        mixer_initialized = True
    except Exception as e:
        print(f"오디오 장치 초기화 실패 (소리 없이 게임이 실행됩니다): {e}")
   
    # 투명 아이콘 설정
    try:
        windowicon = pygame.image.load(DIRCARS + 'icon.png').convert_alpha()
        pygame.display.set_icon(windowicon)
    except:
        pass

    clock = pygame.time.Clock()
   
    # 라이프 표시용 이미지 미리 한 번만 로드하기
    try:
        LIFE_IMAGE = pygame.image.load(DIRCARS + 'Player.png')
        LIFE_IMAGE = pygame.transform.scale(LIFE_IMAGE, (15, 38))
    except:
        LIFE_IMAGE = None

    # 사운드 파일 로드
    if mixer_initialized:
        try:
            pygame.mixer.music.load(DIRSOUND + 'background.ogg')
            pygame.mixer.music.play(-1)
            sound_crash = pygame.mixer.Sound(DIRSOUND + 'crash.ogg')
        except Exception as e:
            print(f"사운드 파일 로드 실패 (파일 확인 필요): {e}")

    # 플레이어 자동차 생성
    player = Car(round(WINDOW_WIDTH / 2), round(WINDOW_HEIGHT - 150), 0, 0)
    player.load_car("p")

    # 자동차 생성
    for i in range(CAR_COUNT):
        car = Car(0, 0, 0, 0)
        car.load_car()
        CARS.append(car)
       
    playing = True

    while playing:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                playing = False
                pygame.quit()
                sys.exit()
               
            if event.type == pygame.KEYDOWN:
                if event.key == pygame.K_RIGHT:
                    player.dx = 5
                elif event.key == pygame.K_LEFT:
                    player.dx = -5
                if event.key == pygame.K_DOWN:
                    player.dy = 5
                elif event.key == pygame.K_UP:
                    player.dy = -5
                   
            if event.type == pygame.KEYUP:
                if event.key in [pygame.K_RIGHT, pygame.K_LEFT]:
                    player.dx = 0
                if event.key in [pygame.K_DOWN, pygame.K_UP]:
                    player.dy = 0
                   
        SCREEN.fill(GRAY)
       
        # 플레이어 업데이트
        player.move_x()
        player.move_y()
        player.check_screen()
        player.draw_car()
       
        # 다른 자동차들 움직이기
        for i in range(CAR_COUNT):
            CARS[i].move_y()
            CARS[i].draw_car()
           
            if CARS[i].rect.y > WINDOW_HEIGHT:
                increase_score()
                CARS[i].load_car()
               
        # 상대 자동차들끼리 충돌 감지
        for i in range(CAR_COUNT):
            for j in range(i + 1, CAR_COUNT):
                if CARS[i].check_collision(CARS[j]):
                    if CARS[i].rect.x > CARS[j].rect.x:
                        CARS[i].rect.x += 4
                        CARS[j].rect.x -= 4
                    else:
                        CARS[i].rect.x -= 4
                        CARS[j].rect.x += 4
                       
                    if CARS[i].rect.y > CARS[j].rect.y:
                        CARS[i].rect.y += CARS[i].dy
                        CARS[j].rect.y -= CARS[j].dy
                    else:
                        CARS[i].rect.y -= CARS[j].dy
                        CARS[j].rect.y += CARS[i].dy
                       
        # 플레이어와 다른 차량 충돌 감지
        for i in range(CAR_COUNT):
            if player.check_collision(CARS[i], 5):
                PNUMBER -= 1
                if sound_crash:  # 이제 안전하게 체크 가능합니다.
                    sound_crash.play()
               
                CARS[i].load_car()
                   
        # 게임 종료 조건
        if PNUMBER <= 0:
            print("Game Over!")
            playing = False

        draw_score()
        pygame.display.flip()
        clock.tick(60)

    pygame.quit()


if __name__ == '__main__':
    main()
