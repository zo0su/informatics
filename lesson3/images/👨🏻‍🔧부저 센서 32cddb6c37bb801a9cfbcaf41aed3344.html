int Buzzer = 7;

void setup() {
  pinMode(Buzzer, OUTPUT);
}

void loop()
{
  tone(Buzzer, 523);
  delay(500);
  tone(Buzzer, 587);
  delay(500);
  tone(Buzzer, 659);
  delay(500);
}
코드 라인설명
int Buzzer = 7;부저의 '+' 핀이 연결된 아두이노의 디지털 7번 핀을 변수로 지정합니다.
pinMode(Buzzer, OUTPUT);7번 핀을 전기를 내보내는 출력 모드로 설정합니다.
tone(Buzzer, 523);부저에 523Hz(도) 주파수의 신호를 보냅니다.
delay(500);0.5초(500ms) 동안 해당 음을 유지하며 기다립니다.

🎼생일 축하합니다. <소스 코드>

int Buzzer = 7;

void setup() {
  pinMode(Buzzer, OUTPUT);
}

void loop() {
  // 1. 생일 축하~ 합니다
  playNote(392, 400); playNote(392, 100); // 솔솔
  playNote(440, 500);                     // 라
  playNote(392, 500);                     // 솔
  playNote(523, 500);                     // 도(높은)
  playNote(494, 1000);                    // 시
  delay(300);

  // 2. 생일 축하~ 합니다
  playNote(392, 400); playNote(392, 100); // 솔솔
  playNote(440, 500);                     // 라
  playNote(392, 500);                     // 솔
  playNote(587, 500);                     // 레(높은)
  playNote(523, 1000);                    // 도(높은)
  delay(300);

  // 3. 사랑하는~ OO의
  playNote(392, 400); playNote(392, 100); // 솔솔
  playNote(784, 500);                     // 솔(아주 높은)
  playNote(659, 500);                     // 미(높은)
  playNote(523, 500);                     // 도(높은)
  playNote(494, 500);                     // 시
  playNote(440, 1000);                    // 라
  delay(300);

  // 4. 생일 축하~ 합니다
  playNote(698, 400); playNote(698, 100); // 파(높은)파(높은)
  playNote(659, 500);                     // 미(높은)
  playNote(523, 500);                     // 도(높은)
  playNote(587, 500);                     // 레(높은)
  playNote(523, 1000);                    // 도(높은)

  delay(5000); // 노래가 끝나고 5초간 대기
}

// 음 재생을 담당하는 사용자 정의 함수
void playNote(int frequency, int duration) {
  tone(Buzzer, frequency, duration);
  delay(duration * 1.1); // 음 길이에 비례해서 약간의 간격을 둡니다
  noTone(Buzzer);        // 확실하게 소리를 끊어줍니다
}

능동 부저

int speakerPin = 8;

void setup()
{
  pinMode(speakerPin, OUTPUT);
}

void loop() {
  digitalWrite(speakerPin, HIGH);
  delay(1000);
  digitalWrite(speakerPin, LOW);
  delay(1000);
}
코드 라인설명
int speakerPin = 8;부저를 아두이노 디지털 8번 핀에 연결하겠다고 선언합니다.
pinMode(speakerPin, OUTPUT);8번 핀을 전기를 보내는 출력용으로 설정합니다.
digitalWrite(speakerPin, HIGH);8번 핀에 5V 전압을 보냅니다. 이때 능동 부저가 "삐~" 소리를 냅니다.
delay(1000);소리를 1초 동안 유지합니다.
digitalWrite(speakerPin, LOW);전압을 0V로 차단합니다. 소리가 멈춥니다.
delay(1000);1초 동안 정적을 유지합니다.
구분능동 부저 (Active Buzzer)피에조/수동 부저 (Passive Buzzer)
내부 구조발진 회로가 내장됨발진 회로가 없음
제어 방법전원만 공급 (HIGH/LOW)주파수 신호 입력 (tone() 함수)
소리 특징단일 음당 고정된 소리 ("삐~")다양한 높낮이 연주 가능 (도레미...)
회로 복잡도매우 간단함상대적으로 복잡함 (신호 제어 필요)
가격상대적으로 비쌈상대적으로 저렴함
주요 용도단순 경보음, 알람, 확인음멜로디 연주, 전자 악기, 정밀 신호음
