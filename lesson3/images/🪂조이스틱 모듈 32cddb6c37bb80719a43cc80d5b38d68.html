🪂조이스틱 모듈

스테퍼 라이브러리 확인

스케치-’라이브러리 관리-Stepper 검색-설치’ 포함에서 확인

#include <Stepper.h> // 스텝 모터 라이브러리 포함

// --- 1. 스텝 모터 설정 (ULN2003 드라이버 기준) ---
// 회전 시킬 스텝 수 (28BYJ-48 모터의 경우 한 바퀴 당 약 2048 스텝)
const int stepsPerRevolution = 2048; 

// 스텝 모터 핀 설정 (IN1, IN3, IN2, IN4 순서로 연결해야 잘 동작하는 경우가 많음)
// 실제 연결한 아두이노 핀 번호로 수정하세요.
Stepper myStepper(stepsPerRevolution, 8, 10, 9, 11);


// --- 2. 조이스틱 설정 ---
int Joy_x = A0; // 좌우 방향 담당 X축
// y, z축은 이 동작에서는 사용하지 않습니다.

// --- 3. 동작 임계값(Threshold) 설정 ---
// 조이스틱의 중앙값은 약 512이지만, 노이즈를 고려하여 
// 중앙 근처에서는 움직이지 않도록 범위를 둡니다.
const int center = 512;
const int threshold = 50; // 이 값보다 더 기울어야 움직임 (예: 512-50 ~ 512+50 범위 밖)


void setup()
{
  // 모터 속도 설정 (RPM) - ULN2003은 너무 빠르면 안됨
  myStepper.setSpeed(5); 
  
  Serial.begin(9600);
}

void loop()
{
  // 1. 조이스틱 X축 값 읽기
  int xValue = analogRead(Joy_x);
  
  Serial.print("X: ");
  Serial.print(xValue);

  // 2. 조이스틱 위치에 따른 모터 구동
  
  // [왼쪽으로 밀었을 때] - 이미지 첫 번째 줄
  if (xValue < (center - threshold)) {
    Serial.println(" -> Moving Left (CCW)");
    // 시계 반대 방향으로 소량 회전 (음수값)
    myStepper.step(-10); 
  }
  // [오른쪽으로 밀었을 때] - 이미지 두 번째 줄
  else if (xValue > (center + threshold)) {
    Serial.println(" -> Moving Right (CW)");
    // 시계 방향으로 소량 회전 (양수값)
    myStepper.step(10); 
  }
  // [중앙에 있을 때]
  else {
    Serial.println(" -> Stopped");
    // 아무 동작 안 함
  }

  // 너무 잦은 시리얼 출력을 방지하기 위한 아주 짧은 대기
  delay(10); 
}
